DECODE ( expression, search, result [, search, result]... [,default] )

SELECT DECODE(NULL, NULL, 1, 0) FROM DUAL;

SELECT first_name, country,
DECODE(country, 'USA', 'North America', 'UK', 'Europe') AS Decode_Result
