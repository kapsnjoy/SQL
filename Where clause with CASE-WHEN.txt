	SELECT *
      FROM Product P
      WHERE (CASE
         WHEN @Status = 'published' THEN
          (CASE
            WHEN P.Status IN (1, 3) THEN
             'TRUE'
            ELSE
             FALSE
          END)
         WHEN @Status = 'standby' THEN
          (CASE
            WHEN P.Status IN (2, 5, 9, 6) THEN
             'TRUE'
            ELSE
             'FALSE'
          END)
         WHEN @Status = 'deleted' THEN
          (CASE
            WHEN P.Status IN (4, 5, 8, 10) THEN
             'TRUE'
            ELSE
             'FALSE'
          END)
         ELSE
          (CASE
            WHEN P.Status IN (1, 3) THEN
             'TRUE'
            ELSE
             'FALSE'
          END)
       END) = 'TRUE'
